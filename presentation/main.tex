\begin{frame}{EV classifier: a frozen LightGBM benchmark}
\begin{columns}[T,onlytextwidth]
\column{.58\textwidth}
\tagline{E22 LIGHTGBM · HELD-OUT TEST}\par\vspace{.25cm}
\small The selected model detects EV-labelled properties from the total net-load profile; it does \alert{not} localise individual charging events.
\vspace{.18cm}
\begin{itemize}\itemsep.18em
  \item 45 features: annual load aggregates, 24-hour mean profile, and windowed multi-meter context.
  \item Selected by validation PR-AUC among 9 candidates; threshold frozen at 0.392 before the test was opened.
  \item Test (67 properties; 15 EV-labelled): ROC-AUC 0.67, recall 67\%, precision 33\%.
  \item Caveat: ``non-EV'' means subsidy-register-unlabelled, not confirmed EV-absent.
\end{itemize}
\column{.37\textwidth}
\centering\textbf{Held-out evaluation}\\[4pt]
\confusiontable{\shortstack[l]{EV-labelled\\(GIGI)}}{\shortstack[l]{Unlabelled\\properties}}{10}{5}{20}{32}\\[5pt]
\scriptsize\textcolor{muted}{Rows are actual labels. $p \geq 0.392$; test $n=67$.}
\end{columns}
\end{frame}

\begin{frame}{Heat-pump boiler: a physical detector, not a classifier}
\begin{columns}[T,onlytextwidth]
\column{.58\textwidth}
\tagline{WPB DETECTOR · LABEL-FREE}\par\vspace{.25cm}
\scriptsize The detector identifies heat-pump water-heater candidates from persistent plateaus in 15-minute grid-import data.
\vspace{.12cm}
\begin{itemize}\itemsep.10em
  \item p10 daily envelope: an event must recur on at least 90\% of days, rejecting irregular loads such as part-time EV charging.
  \item Plateau band: 0.25--1.2 kW for 1.5--9 h; it must recur in both January and July.
  \item Population sweep: 838 of 75,094 households flagged (1.1\%) as candidates.
  \item 4 labelled positives and 0 confirmed negatives: no confusion matrix or accuracy estimate.
\end{itemize}
\column{.37\textwidth}
\centering\textbf{Counterfactual recovery test}\\[3pt]
\metric{79.6\%}{Recovery at 0.3 kW injected}\par\vspace{.18cm}
\metric{94.2\%}{Recovery at 0.5 kW injected}\par\vspace{.18cm}
\small\textcolor{muted}{Synthetic events test sensitivity only; they do not establish precision.}
\end{columns}
\end{frame}

\begin{frame}{Rooftop PV: SolarPrint reads the solar imprint in demand}
\begin{columns}[T,onlytextwidth]
\column{.58\textwidth}
\tagline{HISTGRADIENTBOOSTING · 5-FOLD OOF}\par\vspace{.25cm}
\scriptsize SolarPrint estimates rooftop-PV probability from grid-import shape; feed-in is used offline for silver labels, never as a feature.
\vspace{.12cm}
\begin{itemize}\itemsep.10em
  \item 23 features: summer daytime near-zero import, evening-to-midday contrast, solar coupling, seasonality, and night base load.
  \item Positive--Unlabelled learning with Elkan--Noto correction, calibrated to the 12\% canton base rate.
  \item GIGI result: ROC-AUC 0.92; recall 92\% at $p \geq 0.5$; 6.6\% of the population flagged.
  \item No confirmed non-PV households: the ranking is stronger than absolute probabilities.
\end{itemize}

\begin{frame}{Battery: the model gate is clear; training is next}
\begin{columns}[T,onlytextwidth]
\column{.58\textwidth}
\tagline{PV-CONDITIONED PU DESIGN · NOT YET TRAINED}\par\vspace{.20cm}
\small The intended contrast is battery versus PV-without-battery---not the full population, which would mostly relearn ``has PV.''
\vspace{.12cm}
\begin{itemize}\itemsep.10em
  \item Features: flat import, feed-in shaping, ramp smoothing, evening recovery, and energy balance.
  \item 82 GIGI no-battery households are a held-out audit set, never training negatives.
  \item The export-register gate was corrected: 222/222 usable known batteries are retained.
  \item Training, evaluation, and scoring remain unimplemented: no battery probability or headline metric yet.
\end{itemize}
